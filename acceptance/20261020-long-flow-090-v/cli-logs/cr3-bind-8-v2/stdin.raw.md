import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
const V=path.resolve('acceptance/20261020-long-flow-090-v');
const W=path.join(V,'scenario/东岸末班船-扩展验证');
const R=path.join(W,'.writer/revisions/31b83039-cf67-4d55-9068-b027a090f4ca/workspace');
const unit=Number(process.argv[process.argv.indexOf('--unit')+1]);
if(![8,9,10,11,12].includes(unit)) throw Error('unit');
const hash=b=>crypto.createHash('sha256').update(b).digest('hex');
const read=p=>JSON.parse(fs.readFileSync(p,'utf8'));
const merge=(a,b)=>{for(const[k,v]of Object.entries(b)){if(v&&typeof v==='object'&&!Array.isArray(v))a[k]=merge(a[k]||{},v);else a[k]=v;}return a;};
const input=merge(read(path.join(R,`.writer/drafts/cr3-replay-${unit}-raw.json`)),read(path.join(V,`inputs/cr3-semantic-${unit}.json`)));
const report=`审查/V-C恢复-${String(unit).padStart(4,'0')}-实读重登.md`;
const bytes=fs.readFileSync(path.join(V,`cr3-report-${unit}.md`));
fs.writeFileSync(path.join(R,report),bytes);
const draftSha=hash(fs.readFileSync(path.join(R,input.draft)));
if(draftSha!==input._replay.draft_sha256)throw Error('draft changed since replay-input');
input.replay_review={reviewer:'V同一C实际会话恢复；知情模型实读',report,draft_sha256:draftSha,checked_fields:['summary','characters','timeline','ledger','evidence','downstream']};
input.draft_review={reviewer:'V同一C实际会话恢复；完整章知情自检',report,draft_sha256:draftSha,report_sha256:hash(bytes)};
delete input._instructions;delete input.output;
const out=path.join(V,`inputs/cr3-replay-${unit}-reviewed.json`);
fs.writeFileSync(out,JSON.stringify(input,null,2)+'\n');
console.log(JSON.stringify({unit,out,input_sha256:hash(fs.readFileSync(out)),draft_sha256:draftSha,report,report_sha256:hash(bytes)}));
